// Smart Campus Room Availability Checker -- v1 (single room, monolithic prototype)
// Reconstructed baseline of the Assignment #03 / Practice #06 prototype.
// Acceptance criteria: AC1 valid time, AC2 max 120 min, AC3 blocked room,
// AC4 overlap rules (touching is allowed), AC5 immutable booking data,
// AC6 clear responsive feedback.

var ROOM = { id: 'R101', name: 'Aurora Lab', blocked: false };
var CURRENT_TIME = '09:00';
var BOOKINGS = [
  Object.freeze({ start: '10:00', end: '11:00', title: 'Algorithms lecture' }),
  Object.freeze({ start: '13:00', end: '14:30', title: 'Study group' }),
  Object.freeze({ start: '15:00', end: '16:00', title: 'Robotics club' })
];
Object.freeze(BOOKINGS);

function checkAvailability(start, end) {
  // --- validation (copy A) ---
  var re = /^([01]\d|2[0-3]):([0-5]\d)$/;
  if (!re.test(start) || !re.test(end)) {
    return { status: 'INVALID', message: 'Enter start and end times in HH:MM format (for example 10:30).' };
  }
  var sp = start.split(':');
  var s = parseInt(sp[0], 10) * 60 + parseInt(sp[1], 10);
  var ep = end.split(':');
  var e = parseInt(ep[0], 10) * 60 + parseInt(ep[1], 10);
  var cp = CURRENT_TIME.split(':');
  var c = parseInt(cp[0], 10) * 60 + parseInt(cp[1], 10);
  if (e <= s) {
    return { status: 'INVALID', message: 'End time must be after the start time.' };
  }
  if (s < c) {
    return { status: 'INVALID', message: 'Start time cannot be earlier than the current time (09:00).' };
  }
  if (e - s > 120) {
    return { status: 'TOO_LONG', message: 'A booking can last at most 120 minutes. You requested ' + (e - s) + ' minutes.' };
  }
  if (ROOM.blocked) {
    return { status: 'BLOCKED', message: ROOM.name + ' is blocked and cannot be booked.' };
  }
  for (var i = 0; i < BOOKINGS.length; i++) {
    var bsp = BOOKINGS[i].start.split(':');
    var bs = parseInt(bsp[0], 10) * 60 + parseInt(bsp[1], 10);
    var bep = BOOKINGS[i].end.split(':');
    var be = parseInt(bep[0], 10) * 60 + parseInt(bep[1], 10);
    if (s < be && bs < e) {
      return { status: 'CONFLICT', message: 'Not available: overlaps with "' + BOOKINGS[i].title + '" (' + BOOKINGS[i].start + '\u2013' + BOOKINGS[i].end + ').' };
    }
  }
  return { status: 'AVAILABLE', message: 'Available: ' + ROOM.name + ' is free from ' + start + ' to ' + end + '.' };
}

function describeBooking(b) {
  var a = b.start.split(':');
  var z = b.end.split(':');
  var mins = (parseInt(z[0], 10) * 60 + parseInt(z[1], 10)) - (parseInt(a[0], 10) * 60 + parseInt(a[1], 10));
  return b.start + '\u2013' + b.end + ' ' + b.title + ' (' + mins + ' min)';
}

// --- DOM layer (browser only) ---
function onTimeInput() {
  var start = document.getElementById('start').value;
  var end = document.getElementById('end').value;
  var box = document.getElementById('hint');
  if (!start || !end) { box.innerHTML = ''; return; }
  // --- validation (copy B: live hint duplicates copy A) ---
  var re = /^([01]\d|2[0-3]):([0-5]\d)$/;
  if (!re.test(start) || !re.test(end)) {
    box.innerHTML = '<p class="error">Enter start and end times in HH:MM format (for example 10:30).</p>';
    return;
  }
  var sp = start.split(':');
  var s = parseInt(sp[0], 10) * 60 + parseInt(sp[1], 10);
  var ep = end.split(':');
  var e = parseInt(ep[0], 10) * 60 + parseInt(ep[1], 10);
  if (e <= s) {
    box.innerHTML = '<p class="error">End time must be after the start time.</p>';
    return;
  }
  if (e - s > 120) {
    box.innerHTML = '<p class="error">A booking can last at most 120 minutes. You requested ' + (e - s) + ' minutes.</p>';
    return;
  }
  box.innerHTML = '';
}

function onCheckClick() {
  var result = checkAvailability(document.getElementById('start').value, document.getElementById('end').value);
  var out = document.getElementById('result');
  if (result.status === 'AVAILABLE') {
    out.innerHTML = '<div class="box ok">' + result.message + '</div>';
  } else if (result.status === 'CONFLICT') {
    out.innerHTML = '<div class="box bad">' + result.message + '</div>';
  } else if (result.status === 'BLOCKED') {
    out.innerHTML = '<div class="box bad">' + result.message + '</div>';
  } else {
    out.innerHTML = '<div class="box warn">' + result.message + '</div>';
  }
}

function showBookings() {
  var html = '';
  for (var i = 0; i < BOOKINGS.length; i++) {
    html += '<li>' + describeBooking(BOOKINGS[i]) + '</li>';
  }
  document.getElementById('bookings').innerHTML = html;
}

if (typeof document !== 'undefined') {
  document.getElementById('start').addEventListener('input', onTimeInput);
  document.getElementById('end').addEventListener('input', onTimeInput);
  document.getElementById('check').addEventListener('click', onCheckClick);
  showBookings();
}

if (typeof module !== 'undefined') {
  module.exports = { checkAvailability: checkAvailability, describeBooking: describeBooking, ROOM: ROOM, BOOKINGS: BOOKINGS, CURRENT_TIME: CURRENT_TIME };
}
