function runs(text, base = {}) {
  const out = [];
  text.split(/(\*\*[^*]+\*\*|`[^`]+`)/).forEach((part) => {
    if (!part) return;
    if (part.startsWith('**')) out.push(new TextRun({ text: part.slice(2, -2), bold: true, font: FONT, size: 21, ...base }));
    else if (part.startsWith('`')) out.push(new TextRun({ text: part.slice(1, -1), font: 'Consolas', size: 19, ...base }));
    else out.push(new TextRun({ text: part, font: FONT, size: 21, ...base }));
  });
  return out;
}

  borders, width: { size: width, type: WidthType.DXA },
  shading: o.fill ? { type: ShadingType.CLEAR, fill: o.fill } : undefined,

});
function table(headers, rows, widths, o = {}) {
  const total = widths.reduce((a, b) => a + b, 0);
  return new Table({
    width: { size: total, type: WidthType.DXA }, columnWidths: widths,
    rows: [
      new TableRow({ tableHeader: true, children: headers.map((h, i) => cell(h, widths[i], { fill: 'D5E6E3', bold: true, size: o.size })) }),
      ...rows.map((r) => new TableRow({ cantSplit: true, children: r.map((c, i) => cell(c, widths[i], { size: o.size })) }))
    ]
  });
}

function img(file, width, caption) {
  const dims = { 'before.png': [806, 737], 'after.png': [1684, 838], '01': [1100, 900], '02': [1100, 900], '03': [1100, 900], '04': [375, 1423] };
  const key = file.includes('screenshots') ? file.split('/').pop().slice(0, 2) : file;
  const [w, h] = dims[key];
  const out = [new Paragraph({ alignment: AlignmentType.CENTER, spacing: { before: 80, after: 40 }, keepNext: true,
    children: [new ImageRun({ type: 'png', data: fs.readFileSync(file), transformation: { width, height: Math.round(width * h / w) }, altText: { title: caption, description: caption, name: caption } })] })];

  return out;
}
